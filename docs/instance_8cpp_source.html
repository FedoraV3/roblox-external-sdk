10std::vector<Instance> Instance::getChildren() const {

12
13 const auto start = Memory::rpm<uintptr_t>(Memory::handle(), address + Offsets::Instance::ChildrenStart);
14 const auto end = Memory::rpm<uintptr_t>(Memory::handle(), address + Offsets::Instance::ChildrenEnd);
15 if (!start || !end || *end <= *start)
16 return children;

18 struct shared_ptr_slot {
19 uintptr_t object;
20 uintptr_t ref;
21 };
22
23 const auto count = (*end - *start) / sizeof(shared_ptr_slot);
24 if (count == 0 || count > 0x1000)
25 return children;
26
27 std::vector<shared_ptr_slot> slots(count);
28 if (!ReadProcessMemory(Memory::handle(), reinterpret_cast<LPCVOID>(*start), slots.data(), slots.size() * sizeof(shared_ptr_slot), nullptr))
29 return children;
30
31 children.reserve(count);
32 for (const auto& slot : slots) {
33 if (!slot.object)
34 break;
35
36 children.emplace_back(slot.object);
37 }
38
39 return children;
40}
