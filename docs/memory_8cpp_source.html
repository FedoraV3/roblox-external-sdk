37static Process GetProcessByName(const std::string& processName) {
38 Process proc{};
39 const std::wstring wide(processName.begin(), processName.end());
40
41 DWORD pid = 0;
42 HANDLE snapshot = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
43 if (snapshot != INVALID_HANDLE_VALUE) {
44 PROCESSENTRY32W entry;
45 entry.dwSize = sizeof(PROCESSENTRY32W);
46
47 if (Process32FirstW(snapshot, &entry)) {
48 do {
49 if (_wcsicmp(entry.szExeFile, wide.c_str()) == 0) {
50 pid = entry.th32ProcessID;
51 break;
52 }
53 } while (Process32NextW(snapshot, &entry));
54 }
55 CloseHandle(snapshot);
56 }
57 if (!pid)
58 return proc;
59
60 proc.hProcess = OpenProcess(PROCESS_ALL_ACCESS, FALSE, pid);
61 if (!proc.hProcess)
62 return proc;
63
64 WindowSearch search{pid, nullptr};
65 EnumWindows(FindMainWindow, reinterpret_cast<LPARAM>(&search));
66 proc.hwnd = search.hwnd;
67
68 HANDLE modules = CreateToolhelp32Snapshot(TH32CS_SNAPMODULE | TH32CS_SNAPMODULE32, pid);
69 if (modules != INVALID_HANDLE_VALUE) {
70 MODULEENTRY32W module;
71 module.dwSize = sizeof(MODULEENTRY32W);
72
73 if (Module32FirstW(modules, &module))
74 proc.baseAddress = reinterpret_cast<uintptr_t>(module.modBaseAddr);
75
76 CloseHandle(modules);
77 }
78
79 return proc;
80}

91std::string Memory::readString(uintptr_t addr) {
92 struct sso_t {
93 union {
94 char buffer[0x10];
95 uintptr_t ptr;
96 } string;
97 size_t length;
98 size_t capacity;
99 };
100
101 const auto sso = rpm<sso_t>(hProcess, addr);
102 if (!sso || sso->length == 0 || sso->length > 0x100)
103 return "unknown";
104
105 if (sso->length >= sizeof(sso->string.buffer)) {
106 std::string out(sso->length, '\0');
107 if (!ReadProcessMemory(hProcess, reinterpret_cast<LPCVOID>(sso->string.ptr), out.data(), out.size(), nullptr))
108 return "unknown";
109 return out;
110 }
111
112 return std::string(sso->string.buffer, sso->length);
113}
